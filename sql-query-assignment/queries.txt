## AI Data Retrieval and Ranking Query Design

# Create the Table:
CREATE TABLE STUDENTS (
id Integer Primary Key,
name Text,
score Integer,
passed Boolean,
class Integer
);

# Retrieve all columns from the students table:
SELECT * FROM STUDENTS;

# Retrieve only the name and score columns:
SELECT name, score FROM STUDENTS;

# Retrieve students who scored above 80:
SELECT * FROM STUDENTS WHERE score > 80;

# Retrieve students who scored above 80 AND passed:
SELECT * FROM STUDENTS WHERE score > 80 AND passed = TRUE;

# Retrieve students who scored above 85 OR belong to class 9:
SELECT * FROM STUDENTS WHERE score > 85 OR class = 9;

# Retrieve students who did NOT pass the exam:
SELECT * FROM STUDENTS WHERE NOT passed;

# Return a list of distinct classes:
SELECT DISTINCT class FROM STUDENTS;

# Retrieve the top 5 highest-scoring students:
SELECT * FROM STUDENTS ORDER BY score DESC LIMIT 5;

# Retrieve students sorted first by class (ascending) and then by score (descending):
SELECT * FROM STUDENTS ORDER BY class, score DESC;

# Use an alias to rename score as final_score:
SELECT score as final_score FROM STUDENTS;

# Write one combined query that:
- Selects name and score
- Filters students who scored above 75
- Sorts by score in descending order
- Returns only the top 3 results

SELECT name, score FROM STUDENTS WHERE score > 75 ORDER BY score DESC LIMIT 3;

Query 1: Average salary & employee count (avg > 60000)
SELECT 
    department,
    AVG(salary) AS avg_salary,
    COUNT(emp_id) AS total_employees
FROM employees
GROUP BY department
HAVING AVG(salary) > 60000
ORDER BY avg_salary DESC;

Query 2: Employees working in Mumbai (using JOIN)
SELECT 
    e.name,
    e.salary
FROM employees e
JOIN departments d
ON e.department = d.dept_name
WHERE d.location = 'Mumbai';

Query 3: Highest-paid employee
SELECT 
    name,
    salary
FROM employees
WHERE salary = (SELECT MAX(salary) FROM employees);
